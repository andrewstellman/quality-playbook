# Harness-time runtime dependencies for QPB v1.5.7+.
#
# Install via:
#     pip install -r bin/harness/requirements.txt
#
# The QPB skill itself (what gets bundled into adopter installs via
# bin/install_skill.py) is dependency-free — the skill runs inside arbitrary
# target repos as a Claude / codex / copilot sub-skill, and extra deps would
# pollute the adopter's project. These dependencies are HARNESS-ONLY (the dev
# tooling at bin/qpb_harness.py + bin/harness/**/*.py).
#
# Skill-bundle isolation is enforced by SkillBundleHarnessOnlyDepTests in
# bin/tests/harness/test_platform_compat_180.py.

# textual — interactive TUI for `qpb_harness tui`. Required for the live runs
# dashboard (post-119 rebuild on textual).
textual>=0.50

# psutil — cross-platform process introspection + management. Required since
# 182 (psutil migration replaced ctypes / os.kill probes with psutil.Process
# for pid_alive / kill_process_tree / process_create_time / pid_alive_with_identity
# / wait_for_process). The Windows kill-tree-only-leader bug fix depends on
# psutil's children(recursive=True) walk.
psutil>=5.9

# build — PEP 517 wheel-build front end. Required at harness startup to build
# the QPB skill wheel via `python -m build --wheel` before the orchestrator
# spawns runs. Missing on a fresh dev machine surfaces as
# `ModuleNotFoundError: No module named build` from the wheel-build subprocess.
build>=1.0
